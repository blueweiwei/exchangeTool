JavaScript 核心代码
async function generateSignature(secret, jsonStr, algo) { // ① 解析 JSON，删除 signature 字段 let data = JSON.parse(jsonStr); delete data.signature; // ② keys 按 ASCII 字典序排序 let keys = Object.keys(data).sort(); // ③ 拼接 key=value&key=value... → StringA let pairs = []; for (let k of keys) { let v = data[k]; if (v === null || v === undefined) continue; let strVal = typeof v === 'object' ? JSON.stringify(v) : String(v); pairs.push(k + '=' + strVal); } let stringA = pairs.join('&'); // ④ HMAC 计算（Web Crypto API） let key = await crypto.subtle.importKey( 'raw', TextEncoder(secret), { name: 'HMAC', hash: { name: 'SHA-256' } }, ...); let signBytes = await crypto.subtle.sign('HMAC', key, TextEncoder(stringA)); // ⑤ Base64 编码 → signature let signature = btoa(String.fromCharCode(...new Uint8Array(signBytes))); // ⑥ 写回 JSON data.signature = signature; return { stringA, signature, fullJson: JSON.stringify(data, null, 2) }; }

StringBuilder sb = new StringBuilder(); for (Map.Entry entry : map.entrySet()) { if (sb.length() > 0) sb.append('&'); sb.append(entry.getKey()).append('=').append(entry.getValue()); } String data = sb.toString(); // HMAC-SHA256 SecretKeySpec signingKey = new SecretKeySpec( secret.getBytes(), "HmacSHA256"); Mac mac = Mac.getInstance("HmacSHA256"); mac.init(signingKey); byte[] signBytes = mac.doFinal(data.getBytes("UTF-8")); // Base64 → signature return Base64.getEncoder().encodeToString(signBytes); }

public static String generateSignature(Map<String,String> params, String secretKey) { String baseString = MapUtil.join( new TreeMap<>(params), "&", "=", true); return hmacSHA256Hex(baseString, secretKey); } public static String hmacSHA256Hex(String message, String secretKey) { Mac mac = Mac.getInstance("HmacSHA256"); SecretKeySpec keySpec = new SecretKeySpec( secretKey.getBytes("UTF-8"), "HmacSHA256"); mac.init(keySpec); return bytesToHex(mac.doFinal(message.getBytes("UTF-8"))); } // SHA-256 哈希（用于 bodyHash） public static String sha256Hex(String input) { MessageDigest digest = MessageDigest.getInstance("SHA-256"); return bytesToHex(digest.digest(input.getBytes("UTF-8"))); }
